# Core Loop v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace "type fast until you find a path" with par scoring from an offline shortest-path engine, stuck exits (reset, hint ladder, give up), a tap-to-close win moment, a best-route comparison on results, and Play Again through the reveal.

**Architecture:** `npm run build:graph` writes `data/costar-graph.json` (pool actors' credits plus the top-15 casts of titles with 100+ votes). Pure solver functions in `lib/solver/` answer shortest-route and puzzle-picking questions in microseconds. `/api/puzzle` deals verified pairs with par. `/api/route` serves hints and give-up routes, bridging off-graph actors through TMDb. Player moves keep going through the unchanged `/api/tmdb/validate`. The reducer carries par, hints, pause time, closing and end reason.

**Tech Stack:** Next.js 16 App Router, React 19 useReducer, TypeScript, Tailwind v4, Node 22 test runner via `npx tsx --test` (no new dependencies).

**Spec:** `docs/superpowers/specs/2026-09-25-core-loop-v2-design.md`

## Global Constraints

- Every eligibility decision goes through `lib/tmdb-rules.ts`. Never filter credits inline.
- Difficulty is `"easy" | "medium"` only: easy = par 1, medium = par 2.
- No unverified pair may reach `START_GAME`. There is no fallback pair anywhere.
- Score = steps + hintsUsed − par. Time is a tiebreaker only and excludes paused validation time.
- The reducer is pure: time comes in on actions as `now`, and there is no `Date.now()` inside `gameReducer`.
- Win motion values live in `lib/motion.ts` and nowhere else.
- No visual redesign. New UI reuses the existing tokens and classes.
- Commit after each numbered build item, on branch `core-loop-v2`. Commit trailer:
  `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>` and `Claude-Session: https://claude.ai/code/session_01QRTL4Grp9iXacXLGDDPnjh`

## Review Focus

1. **Share link to a pair with no route, or ids outside the pool.** The player should see a clear dead-end screen with a New puzzle button, never a silent random pair. → Task 4 route test plus the Task 5 play page.
2. **A validation call fails mid-pause.** The timer must resume and must not stay frozen forever. → Task 6 reducer test (RESUME without PAUSE is a no-op; PAUSE twice keeps the first start) plus `finally` in ChainBuilder.
3. **Hint requested from an actor outside the graph** (the player used an obscure film). The bridge should find a route or answer "no hint from here" at zero cost. → Task 4 bridge tests (off-graph success, off-graph failure).
4. **Player types the target's name in search instead of tapping the card.** It should close the chain exactly like a tap and never jump straight to results. → Task 8 reducer test (`SELECT_PERSON` with the target id is a no-op) plus the ChainBuilder routing.
5. **Actions arriving in the wrong phase** (double-tap on the target, FINISH without closing, GIVE_UP after closing). The state must stay unchanged. → Task 7/8 reducer guard tests.

```ts
// lib/solver/graph.ts
import type { Graph, GraphFile } from "./types";

export function loadGraph(file: GraphFile): Graph {
  const actorTitles: number[][] = file.actors.map(() => []);
  const titleActors = file.cast.map((members, t) => {
    const sorted = [...new Set(members)].sort((a, b) => a - b);
    for (const a of sorted) actorTitles[a].push(t);
    return sorted;
  });
  return {
    actorIds: file.actors.map(([id]) => id),
    actorNames: file.actors.map(([, name]) => name),
    titles: file.titles.map(([id, name, mediaType, year, votes]) => ({ id, name, mediaType, year, votes })),
    actorTitles,
    titleActors,
    actorIndex: new Map(file.actors.map(([id], i) => [id, i])),
  };
}
```

```ts
// lib/solver/search.ts
// Bidirectional BFS over the actor graph, where two actors are adjacent when they
// share a title. One "step" is one shared title, so par = number of titles.
import type { Graph, Route, RouteStep } from "./types";

type Parent = { prev: number; via: number } | null; // actor index + title index

function expand(g: Graph, frontier: number[], seen: Map<number, Parent>): number[] {
  const next: number[] = [];
  for (const a of frontier) {
    for (const t of g.actorTitles[a]) {
      for (const b of g.titleActors[t]) {
        if (seen.has(b)) continue;
        seen.set(b, { prev: a, via: t });
        next.push(b);
      }
    }
  }
  return next.sort((x, y) => x - y);
}

function depthOf(seen: Map<number, Parent>, a: number): number {
  let d = 0;
  for (let p = seen.get(a); p; p = seen.get(p.prev)) d++;
  return d;
}

export function shortestRoute(g: Graph, fromId: number, toId: number): Route | null {
  const from = g.actorIndex.get(fromId);
  const to = g.actorIndex.get(toId);
  if (from === undefined || to === undefined) return null;
  const actor = (i: number): RouteStep => ({ kind: "actor", id: g.actorIds[i], name: g.actorNames[i] });
  if (from === to) return { steps: [actor(from)], par: 0 };

  const fSeen = new Map<number, Parent>([[from, null]]);
  const bSeen = new Map<number, Parent>([[to, null]]);
  let fFront = [from];
  let bFront = [to];

  while (fFront.length && bFront.length) {
    const forward = fFront.length <= bFront.length;
    const [seen, other] = forward ? [fSeen, bSeen] : [bSeen, fSeen];
    const next = expand(g, forward ? fFront : bFront, seen);
    // Every meeting found in this level: pick the shortest total, then the lowest actor index.
    let best: { meet: number; len: number } | null = null;
    for (const m of next) {
      if (!other.has(m)) continue;
      const len = depthOf(fSeen, m) + depthOf(bSeen, m);
      if (!best || len < best.len || (len === best.len && m < best.meet)) best = { meet: m, len };
    }
    if (best) return build(g, fSeen, bSeen, best.meet, actor);
    if (forward) fFront = next; else bFront = next;
  }
  return null;
}

function build(
  g: Graph, fSeen: Map<number, Parent>, bSeen: Map<number, Parent>, meet: number,
  actor: (i: number) => RouteStep,
): Route {
  const title = (t: number): RouteStep => {
    const x = g.titles[t];
    return { kind: "title", id: x.id, name: x.name, mediaType: x.mediaType, year: x.year };
  };
  const left: RouteStep[] = [actor(meet)];
  for (let p = fSeen.get(meet); p; p = fSeen.get(p.prev)) left.unshift(actor(p.prev), title(p.via));
  const steps = left;
  for (let p = bSeen.get(meet); p; p = bSeen.get(p.prev)) steps.push(title(p.via), actor(p.prev));
  return { steps, par: (steps.length - 1) / 2 };
}
```

Note on the tie-break in the fixture: from A, the forward level 1 is {B(1), F(5)} (sorted), and backward from C, level 1 is {B(1), D(3), F(5)}. The smaller frontier expands first (forward, size 1). B and F both meet at total length 2, so B wins on the lower index, which gives A→AB→B→BC→C.

```ts
// lib/solver/puzzle.ts
import type { Graph, SolverDifficulty } from "./types";
import { shortestRoute } from "./search";

export const PAR_FOR: Record<SolverDifficulty, number> = { easy: 1, medium: 2 };

// Uniform pick of two different ids (replaces the biased sort-shuffle).
export function sampleDistinctPair(ids: number[], rng: () => number): [number, number] {
  const i = Math.floor(rng() * ids.length) % ids.length;
  let j = Math.floor(rng() * (ids.length - 1)) % (ids.length - 1);
  if (j >= i) j++;
  return [ids[i], ids[j]];
}

export function pickPuzzle(
  g: Graph, poolIds: number[], difficulty: SolverDifficulty,
  rng: () => number = Math.random, maxSamples = 200,
): { startId: number; targetId: number; par: number } | null {
  const ids = poolIds.filter((id) => g.actorIndex.has(id));
  if (ids.length < 2) return null;
  const want = PAR_FOR[difficulty];
  for (let n = 0; n < maxSamples; n++) {
    const [startId, targetId] = sampleDistinctPair(ids, rng);
    const r = shortestRoute(g, startId, targetId);
    if (r?.par === want) return { startId, targetId, par: want };
  }
  return null;
}
```

```ts
//
//   npm run build:graph        # writes data/costar-graph.json
//
// Graph = every eligible acting credit of every pool actor, plus the top 15
// billed cast (acting roles only) of each of those titles with 100+ votes.
// Non-pool actors who appear in only one title are dropped: they can never be
// in the middle of a route. Responses are cached in .cache/graph-build so a
// rerun only fetches what changed.
import { readFileSync, writeFileSync, existsSync, mkdirSync } from "node:fs";
import { createHash } from "node:crypto";
import { resolve, join } from "node:path";
import {
  TMDB_BASE, castMemberActs, castPath, creditKey, isActingRole, isAwardsShow, toMediaType,
  type CastMember, type CreditEntry,
} from "../lib/tmdb-rules";
import { loadGraph } from "../lib/solver/graph";
import { shortestRoute } from "../lib/solver/search";
import type { GraphFile } from "../lib/solver/types";

const CAST_DEPTH = 15;
const MIN_VOTES = 100;
const CONCURRENCY = 10;
const root = resolve(__dirname, "..");
const cacheDir = join(root, ".cache", "graph-build");
mkdirSync(cacheDir, { recursive: true });

function apiKey(): string {
  if (process.env.TMDB_API_KEY) return process.env.TMDB_API_KEY;
  const env = readFileSync(resolve(root, ".env.local"), "utf8");
  const m = env.match(/^TMDB_API_KEY=["']?([^"'\n]+)/m);
  if (!m) throw new Error("TMDB_API_KEY not found in env or .env.local");
  return m[1];
}
const KEY = apiKey();
let network = 0;

async function tmdb(path: string): Promise<any> {
  const file = join(cacheDir, createHash("sha1").update(path).digest("hex") + ".json");
  if (existsSync(file)) return JSON.parse(readFileSync(file, "utf8"));
  const url = `${TMDB_BASE}${path}${path.includes("?") ? "&" : "?"}api_key=${KEY}`;
  for (let attempt = 0; attempt < 5; attempt++) {
    const res = await fetch(url).catch(() => null);
    if (res?.ok) { network++; const data = await res.json(); writeFileSync(file, JSON.stringify(data)); return data; }
    if (res?.status === 404) { writeFileSync(file, "null"); return null; }
    await new Promise((r) => setTimeout(r, 750 * (attempt + 1)));
  }
  throw new Error(`TMDb request failed: ${path}`);
}

async function mapLimit<T, R>(items: T[], limit: number, fn: (item: T) => Promise<R>): Promise<R[]> {
  const out: R[] = new Array(items.length);
  let next = 0;
  await Promise.all(Array.from({ length: limit }, async () => {
    while (next < items.length) { const i = next++; out[i] = await fn(items[i]); }
  }));
  return out;
}

type Credit = CreditEntry & { title?: string; name?: string; release_date?: string; first_air_date?: string };
type Title = { key: string; id: number; mediaType: "movie" | "tv"; name: string; year: string; votes: number; pool: Set<number> };

async function main() {
  const t0 = Date.now();
  const pool: { id: number; name: string }[] = JSON.parse(readFileSync(join(root, "data/actor-pool.json"), "utf8")).actors;
  const poolIds = new Set(pool.map((a) => a.id));

  // 1. Every eligible acting credit of every pool actor.
  const titles = new Map<string, Title>();
  const credits = await mapLimit(pool, CONCURRENCY, (a) => tmdb(`/person/${a.id}/combined_credits`));
  credits.forEach((data, i) => {
    for (const c of (data?.cast ?? []) as Credit[]) {
      if (!isActingRole(c)) continue;
      const mediaType = toMediaType(c.media_type);
      const name = c.title ?? c.name ?? "";
      if (isAwardsShow(mediaType, name)) continue;
      const key = creditKey(c);
      const t = titles.get(key) ?? {
        key, id: c.id, mediaType, name,
        year: (c.release_date ?? c.first_air_date ?? "").slice(0, 4),
        votes: c.vote_count ?? 0, pool: new Set<number>(),
      };
      t.pool.add(pool[i].id);
      titles.set(key, t);
    }
  });

  // 2. Casts. Titles under MIN_VOTES only keep their pool actors (no fetch needed).
  const all = [...titles.values()];
  const big = all.filter((t) => t.votes >= MIN_VOTES);
  const casts = await mapLimit(big, CONCURRENCY, (t) => tmdb(castPath(t.mediaType, t.id)));
  const members = new Map<string, { id: number; name: string }[]>();
  big.forEach((t, i) => {
    const cast = ((casts[i]?.cast ?? []) as (CastMember & { name: string; order?: number })[])
      .filter(castMemberActs)
      .slice(0, CAST_DEPTH)
      .map((m) => ({ id: m.id, name: m.name }));
    members.set(t.key, cast);
  });

  // 3. Assemble, then drop non-pool actors with a single title.
  const names = new Map(pool.map((a) => [a.id, a.name]));
  const titleCast = all.map((t) => {
    const ids = new Set<number>(t.pool);
    for (const m of members.get(t.key) ?? []) { ids.add(m.id); if (!names.has(m.id)) names.set(m.id, m.name); }
    return [...ids];
  });
  const degree = new Map<number, number>();
  for (const ids of titleCast) for (const id of ids) degree.set(id, (degree.get(id) ?? 0) + 1);
  const keep = (id: number) => poolIds.has(id) || (degree.get(id) ?? 0) >= 2;

  const actorIds = [...degree.keys()].filter(keep).sort((a, b) => a - b);
  const actorIdx = new Map(actorIds.map((id, i) => [id, i]));
  const kept = all.map((t, i) => ({ t, ids: titleCast[i].filter(keep) })).filter((x) => x.ids.length >= 2);

  const file: GraphFile = {
    builtAt: new Date().toISOString(),
    params: { castDepth: CAST_DEPTH, minVotes: MIN_VOTES },
    actors: actorIds.map((id) => [id, names.get(id) ?? ""]),
    titles: kept.map(({ t }) => [t.id, t.name, t.mediaType, t.year, t.votes]),
    cast: kept.map(({ ids }) => ids.map((id) => actorIdx.get(id)!).sort((a, b) => a - b)),
  };

  // 4. Checks: every pool actor present; par histogram over all pool pairs.
  const missing = pool.filter((a) => !actorIdx.has(a.id));
  if (missing.length) throw new Error(`Pool actors missing from graph: ${missing.map((a) => a.name).join(", ")}`);
  const g = loadGraph(file);
  const hist: Record<string, number> = {};
```

```ts
import type { Graph, GraphMediaType, Route, RouteStep } from "./types";
import { shortestRoute } from "./search";

export interface BridgeTitle { id: number; name: string; mediaType: GraphMediaType; year: string; votes: number }
export interface CreditsSource {
  titlesFor(actorId: number): Promise<BridgeTitle[]>;
  castOf(title: BridgeTitle): Promise<number[]>;
}

// A player can reach actors the graph pruned away (obscure films are allowed and
// can beat par). To hint from there, look at that actor's biggest titles and
// hop onto the first graph actor with the shortest onward route.
export async function routeFromAnyActor(
  g: Graph, fromId: number, toId: number, src: CreditsSource,
  { fromName = "", maxTitles = 10 }: { fromName?: string; maxTitles?: number } = {},
): Promise<Route | null> {
  if (!g.actorIndex.has(toId)) return null;
  if (g.actorIndex.has(fromId)) return shortestRoute(g, fromId, toId);

  const titles = (await src.titlesFor(fromId)).sort((a, b) => b.votes - a.votes).slice(0, maxTitles);
  const casts = await Promise.all(titles.map((t) => src.castOf(t)));
  let best: { title: BridgeTitle; onward: Route } | null = null;
  titles.forEach((title, i) => {
    for (const id of [...new Set(casts[i])].sort((a, b) => a - b)) {
      if (id === fromId || !g.actorIndex.has(id)) continue;
      const onward = shortestRoute(g, id, toId);
      if (onward && (!best || onward.par < best.onward.par)) best = { title, onward };
    }
  });
  if (!best) return null;
  const { title, onward } = best as { title: BridgeTitle; onward: Route };
  const head: RouteStep[] = [
    { kind: "actor", id: fromId, name: fromName },
    { kind: "title", id: title.id, name: title.name, mediaType: title.mediaType, year: title.year },
  ];
  return { steps: [...head, ...onward.steps], par: onward.par + 1 };
}
```

```ts
export async function eligibleCredits(apiKey: string, actorId: number): Promise<Credit[]> {
  const res = await fetch(`${TMDB_BASE}/person/${actorId}/combined_credits?api_key=${apiKey}`, tmdbCache(TMDB_CREDITS_TTL));
  if (!res.ok) throw new Error(`combined_credits ${res.status}`);
  const seen = new Set<string>();
  return ((await res.json()).cast as Credit[] ?? []).filter((c) => {
    const key = creditKey(c);
    if (seen.has(key) || !isActingRole(c) || isAwardsShow(toMediaType(c.media_type), c.title ?? c.name)) return false;
    seen.add(key);
    return true;
  });
}

export function tmdbCreditsSource(apiKey: string): CreditsSource {
  return {
    async titlesFor(actorId) {
      return (await eligibleCredits(apiKey, actorId)).map((c): BridgeTitle => ({

        year: (c.release_date ?? c.first_air_date ?? "").slice(0, 4), votes: c.vote_count ?? 0,

      const res = await fetch(`${TMDB_BASE}${castPath(t.mediaType, t.id)}?api_key=${apiKey}`, tmdbCache(TMDB_CREDITS_TTL));
      if (!res.ok) return [];
```

```ts
import { DAY, cachedJson, rateLimit } from "@/lib/api-cache";
import { getGraph, getPool } from "@/lib/solver/server";

// Deals a verified pair with its par. Never falls back to an unchecked pair.
export async function GET(request: NextRequest) {
  const limited = rateLimit(request);
  if (limited) return limited;
  const params = new URL(request.url).searchParams;
  let graph, pool;
  try { graph = getGraph(); pool = getPool(); } catch {
    return NextResponse.json({ error: "Puzzle data unavailable" }, { status: 503 });
  }
  const byId = new Map(pool.map((a) => [a.id, a]));

  const start = Number(params.get("start"));
  const target = Number(params.get("target"));
  if (params.has("start") || params.has("target")) {
    const s = byId.get(start), t = byId.get(target);
    const par = s && t && start !== target ? shortestRoute(graph, start, target)?.par : undefined;
    if (!s || !t || !par) return NextResponse.json({ error: "This pair can't be played" }, { status: 404 });
    return cachedJson({ start: s, target: t, par }, DAY);
  }

  const difficulty = params.get("difficulty");
  if (difficulty !== "easy" && difficulty !== "medium") {
    return NextResponse.json({ error: "difficulty must be easy or medium" }, { status: 400 });
  }
  const pick = pickPuzzle(graph, pool.map((a) => a.id), difficulty);
  if (!pick) return NextResponse.json({ error: "No puzzle found" }, { status: 503 });
  return NextResponse.json(
    { start: byId.get(pick.startId), target: byId.get(pick.targetId), par: pick.par },
    { headers: { "Cache-Control": "no-store" } },
  );
}
```
